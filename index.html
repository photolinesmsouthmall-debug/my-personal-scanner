function parseText(text) { // 1. RECEIPT #: Matches # followed by 6-10 digits (#0000111022) const receiptMatch = text.match(/#\s*(\d{6,10})/); if (receiptMatch) { const receiptNo = receiptMatch[0]; if (scannedReceipts.has(receiptNo)) { statusText.innerText = `Skipped Duplicate: ${receiptNo}`; statusText.style.color = "#ffcc00"; return; } // 2. CUSTOMER NAME: Flexibly handles "SN#", "SNH", or "SN" followed by code, dashes, or colons let customerName = "UNKNOWN"; // Matches "SNH:B08903 - CARLOS", "SN#:B08901 CARLOS", or "SN:B08903-CARLOS" const snPattern = /(?:SN[#H]?|S\/N)\s*[:.-]?\s*[A-Z0-9]+\s*[-:\s]+\s*([A-Za-z\s]{3,20})/i; const match = text.match(snPattern); if (match && match[1]) { customerName = match[1].trim(); } else { // Fallback: look for any name following a hyphen/dash after an alphanumeric code const dashMatch = text.match(/[A-Z0-9]{5,8}\s*-\s*([A-Za-z\s]{3,20})/i); if (dashMatch && dashMatch[1]) { customerName = dashMatch[1].trim(); } } // 3. AMOUNT: Explicitly extracts from SUB-TOTAL to avoid grabbing unit prices let amount = 0.00; // Priority 1: Match line with SUB-TOTAL const subTotalMatch = text.match(/SUB-?\s*TOTAL\s*[:=]?\s*([\d,]+\.\d{2})/i); if (subTotalMatch && subTotalMatch[1]) { amount = parseFloat(subTotalMatch[1].replace(',', '')); } else { // Priority 2: Grab the right-most number on the SUB-TOTAL line if OCR splits it const lines = text.split('\n'); for (let line of lines) { if (/SUB-?\s*TOTAL/i.test(line)) { const lineAmounts = line.match(/[\d,]+\.\d{2}/g); if (lineAmounts && lineAmounts.length > 0) { amount = parseFloat(lineAmounts[lineAmounts.length - 1].replace(',', '')); break; } } } } // Fallback if SUB-TOTAL is missed completely: grab bottom-most decimal amount if (amount === 0.00) { const allAmounts = text.match(/[\d,]+\.\d{2}/g); if (allAmounts && allAmounts.length > 0) { amount = parseFloat(allAmounts[allAmounts.length - 1].replace(',', '')); } } // Save scan and log balance scannedReceipts.add(receiptNo); customerBalances[customerName] = (customerBalances[customerName] || 0) + amount; logData(receiptNo, customerName, amount); statusText.innerText = `Scanned ${receiptNo}! Total for ${customerName}: ₱${amount.toFixed(2)}`; statusText.style.color = "#00ff00"; } }
